(function(){ const form = document.getElementById('pg-settings-form'); const button = document.getElementById('pg-test-btn'); const result = document.getElementById('pg-test-result'); button.addEventListener('click', async function(){ if (!form.reportValidity()) return; button.disabled = true; button.textContent = 'در حال بررسی…'; result.hidden = true; try { const data = new FormData(form); data.set('action', 'test'); data.set('ajax', '1'); const response = await fetch(form.getAttribute('action'), {method: 'POST', body: data, credentials: 'same-origin', headers: {'X-Requested-With': 'XMLHttpRequest'}}); const payload = await response.json(); result.className = 'flash ' + (payload.ok ? 'ok' : 'err'); result.textContent = payload.message || payload.error || 'تست اتصال ناموفق بود'; } catch (_) { result.className = 'flash err'; result.textContent = 'تست اتصال ناموفق بود؛ دوباره تلاش کنید.'; } finally { result.hidden = false; button.disabled = false; button.textContent = 'تست اتصال'; } }); })();
